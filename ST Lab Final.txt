it is redirected to google.co.in


package selenium;

import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class program1 {

public static void main(String[] args) {


WebDriver driver = new ChromeDriver();

        driver.get("https://www.google.com");

        String Title = driver.getTitle();

        if (Title.contains("google")) {

            System.out.println("Title Verification Passed");

        } else {

            System.out.println("Title Verification Failed");

        }

        String currentUrl = driver.getCurrentUrl();

        if (currentUrl.contains("google.com")) {

            System.out.println("Redirected to google.com - Passed");

        } else {

            System.out.println("Redirected URL: " + currentUrl);

            System.out.println("Redirection Verification Failed");

        }

        driver.quit();

}


}

5. Write a script to search for specified option in the listbox

package selenium;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.Select;

public class program5 {

public static void main(String[] args) {

WebDriver driver = new ChromeDriver();
     driver.get("https://demoqa.com/select-menu");

        driver.manage().window().maximize();

        WebElement listBox = driver.findElement(By.id("oldSelectMenu"));

        Select select = new Select(listBox);

        String searchOption = "Purple"; 

        boolean found = false;
        for (WebElement option : select.getOptions()) {
            if (option.getText().equalsIgnoreCase(searchOption)) {
                found = true;
                option.click();   
                System.out.println(searchOption + " found and selected.");

                break;

            }
        }

        if (!found) {
            System.out.println(searchOption + " not found in the list box.");

        }
        driver.quit();
}

}

6. Write a script to print the content of list in sorted order

package selenium;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.Select;

public class program6 {
public static void main(String[] args) {

        WebDriver driver = new ChromeDriver();
        driver.get("https://demoqa.com/select-menu");
        driver.manage().window().maximize();
        WebElement dropdown = driver.findElement(By.id("oldSelectMenu"));
        Select select = new Select(dropdown);
        List<WebElement> options = select.getOptions();
        ArrayList<String> colors = new ArrayList<>();
        for (WebElement option : options) {
            colors.add(option.getText());
        }
        Collections.sort(colors);
        System.out.println("Contents of the list in sorted order:");
        for (String color : colors) {
            System.out.println(color);
        }

        driver.quit();

}

}

8. Write a script to close all the browsers without using quit() method.

package selenium;





import java.util.Set;



import org.openqa.selenium.WebDriver;

import org.openqa.selenium.chrome.ChromeDriver;

public class program7 {



    public static void main(String[] args) throws InterruptedException {



        // Launch browser

        WebDriver driver = new ChromeDriver();



        // Open first website

        driver.get("https://www.google.com");



        // Open second window

        driver.switchTo().newWindow(org.openqa.selenium.WindowType.WINDOW);

        driver.get("https://www.amazon.in");



        // Open third window

        driver.switchTo().newWindow(org.openqa.selenium.WindowType.WINDOW);

        driver.get("https://www.flipkart.com");



        // Get all window handles

        Set<String> windows = driver.getWindowHandles();



        // Close each browser window

        for (String window : windows) {

            driver.switchTo().window(window);

            driver.close();

        }

    }

}

9.Write a script to handle all locators and return web element for any locator

package selenium;



import java.time.Duration;

import org.openqa.selenium.By;

import org.openqa.selenium.WebDriver;

import org.openqa.selenium.WebElement;

import org.openqa.selenium.support.ui.ExpectedConditions;

import org.openqa.selenium.support.ui.WebDriverWait;

import org.openqa.selenium.chrome.ChromeDriver;



public class program9 {



    public static WebElement getElement(WebDriver driver, String locatorType, String locatorValue) {



        By by;



        switch(locatorType.toLowerCase()) {



            case "id":

                by = By.id(locatorValue);

                break;



            case "name":

                by = By.name(locatorValue);

                break;



            case "classname":

                by = By.className(locatorValue);

                break;



            case "tagname":

                by = By.tagName(locatorValue);

                break;



            case "linktext":

                by = By.linkText(locatorValue);

                break;



            case "partiallinktext":

                by = By.partialLinkText(locatorValue);

                break;



            case "css":

            case "cssselector":

                by = By.cssSelector(locatorValue);

                break;



            case "xpath":

                by = By.xpath(locatorValue);

                break;



            default:

                throw new IllegalArgumentException("Invalid Locator Type");

        }



        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));



        return wait.until(ExpectedConditions.visibilityOfElementLocated(by));

    }

    

    public static void main(String[] args) {



        WebDriver driver = new ChromeDriver();



        driver.manage().window().maximize();



        driver.get("https://www.amazon.com");



        WebElement searchBox = getElement(driver, "id", "twotabsearchtextbox");



        System.out.println("WebElement Returned Successfully");

        System.out.println("Tag Name : " + searchBox.getTagName());

        System.out.println("Element ID : " + searchBox.getAttribute("id"));



        searchBox.sendKeys("Laptop");



        WebElement searchButton = getElement(driver, "id", "nav-search-submit-button");

        searchButton.click();



        driver.quit();

    }

}

14.Write a script to access a link in selenium web driver by linktext() and
partialLinktext()

package selenium;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class program14 {

    public static void main(String[] args) {

        // Launch Chrome browser
        WebDriver driver = new ChromeDriver();

        // Open the website
        driver.get("https://the-internet.herokuapp.com/");

        // Access link using linkText()
        driver.findElement(By.linkText("A/B Testing")).click();

        // Check whether A/B Testing page opened
        if (driver.getCurrentUrl().contains("/abtest")) {
            System.out.println("A/B Testing link opened successfully");
        } else {
            System.out.println("A/B Testing link failed");
        }

        // Go back to the previous page
        driver.navigate().back();

        // Access link using partialLinkText()
        driver.findElement(By.partialLinkText("Form")).click();

        // Check whether Form Authentication page opened
        if (driver.getCurrentUrl().contains("/login")) {
            System.out.println("Form Authentication link opened successfully");
        } else {
            System.out.println("Form Authentication link failed");
        }

        // Close the browser
        driver.quit();

        System.out.println("Browser closed successfully");
    }
}

15. Write a script to Locate a Link by Selecting Multiple Items in a Dropdown

package selenium;



import org.openqa.selenium.By;

import org.openqa.selenium.WebDriver;

import org.openqa.selenium.WebElement;

import org.openqa.selenium.edge.EdgeDriver;

import org.openqa.selenium.support.ui.Select;



public class program15 {



    public static void main(String[] args) {



        // Launch Edge browser

        WebDriver driver = new EdgeDriver();



        // Open the Selenium form page

        driver.get("https://www.selenium.dev/selenium/web/formPage.html");



        // Locate the multi-select dropdown

        Select dropdown = new Select(

                driver.findElement(By.name("multi-select"))

        );



        // Check whether dropdown supports multiple selection

        if (dropdown.isMultiple()) {



            // Select multiple items by index

            dropdown.selectByIndex(0);

            dropdown.selectByIndex(1);



            // Get selected items

            java.util.List<WebElement> selectedItems =

                    dropdown.getAllSelectedOptions();



            // Check whether multiple items are selected

            if (selectedItems.size() == 2) {

                System.out.println("Multiple items selected successfully");

            } else {

                System.out.println("Multiple items selection failed");

            }



        } else {

            System.out.println("Dropdown does not support multiple selection");

        }



        // Close browser

        driver.quit();



        System.out.println("Browser closed successfully");

    }

}
